#include <stdio.h>

#define MAX 100
#define MAXV 20
#define MAXCAP 1000

/* ---------- 1. SELECTION SORT ---------- */
void selectionSort(void) {
    int n, a[MAX], i, j, min, temp;

    printf("Enter number of elements: ");
    scanf("%d", &n);
    printf("Enter elements: ");
    for (i = 0; i < n; i++) scanf("%d", &a[i]);

    for (i = 0; i < n - 1; i++) {
        min = i;
        for (j = i + 1; j < n; j++)
            if (a[j] < a[min]) min = j;

        temp = a[i]; a[i] = a[min]; a[min] = temp;
    }

    printf("Sorted array: ");
    for (i = 0; i < n; i++) printf("%d ", a[i]);
    printf("\n");
}

/* ---------- 2. INSERTION SORT ---------- */
void insertionSort(void) {
    int n, a[MAX], i, j, key;

    printf("Enter number of elements: ");
    scanf("%d", &n);
    printf("Enter elements: ");
    for (i = 0; i < n; i++) scanf("%d", &a[i]);

    for (i = 1; i < n; i++) {
        key = a[i];
        j = i - 1;
        while (j >= 0 && a[j] > key) {
            a[j + 1] = a[j];
            j--;
        }
        a[j + 1] = key;
    }

    printf("Sorted array: ");
    for (i = 0; i < n; i++) printf("%d ", a[i]);
    printf("\n");
}

/* ---------- 3. QUICK SORT ---------- */
void quickSortArray(int a[], int low, int high) {
    int pivot, i, j, temp;

    if (low >= high) return;

    pivot = a[high];
    i = low - 1;

    for (j = low; j < high; j++) {
        if (a[j] < pivot) {
            i++;
            temp = a[i]; a[i] = a[j]; a[j] = temp;
        }
    }

    temp = a[i + 1]; a[i + 1] = a[high]; a[high] = temp;

    quickSortArray(a, low, i);
    quickSortArray(a, i + 2, high);
}

void quickSort(void) {
    int n, a[MAX], i;

    printf("Enter number of elements: ");
    scanf("%d", &n);
    printf("Enter elements: ");
    for (i = 0; i < n; i++) scanf("%d", &a[i]);

    quickSortArray(a, 0, n - 1);

    printf("Sorted array: ");
    for (i = 0; i < n; i++) printf("%d ", a[i]);
    printf("\n");
}

/* ---------- 4. BINARY SEARCH ---------- */
void binarySearch(void) {
    int n, a[MAX], i, key, low, high, mid, found = -1;

    printf("Enter number of elements (sorted array): ");
    scanf("%d", &n);
    printf("Enter sorted elements: ");
    for (i = 0; i < n; i++) scanf("%d", &a[i]);
    printf("Enter element to search: ");
    scanf("%d", &key);

    low = 0;
    high = n - 1;

    while (low <= high) {
        mid = (low + high) / 2;
        if (a[mid] == key) {
            found = mid;
            break;
        } else if (a[mid] < key) {
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }

    if (found != -1) printf("Element found at position %d\n", found + 1);
    else printf("Element not found\n");
}

/* ---------- 5. FRACTIONAL KNAPSACK ---------- */
void fractionalKnapsack(void) {
    int n, p[MAX], w[MAX], capacity, i, j, temp;
    double ratio[MAX], r, profit = 0.0;

    printf("Enter number of items: ");
    scanf("%d", &n);
    printf("Enter profit and weight of each item:\n");
    for (i = 0; i < n; i++) {
        scanf("%d%d", &p[i], &w[i]);
        if (w[i] <= 0) {
            printf("Weights must be positive.\n");
            return;
        }
        ratio[i] = (double)p[i] / w[i];
    }

    printf("Enter knapsack capacity: ");
    scanf("%d", &capacity);

    for (i = 0; i < n - 1; i++) {
        for (j = i + 1; j < n; j++) {
            if (ratio[i] < ratio[j]) {
                r = ratio[i]; ratio[i] = ratio[j]; ratio[j] = r;
                temp = p[i]; p[i] = p[j]; p[j] = temp;
                temp = w[i]; w[i] = w[j]; w[j] = temp;
            }
        }
    }

    for (i = 0; i < n && capacity > 0; i++) {
        if (capacity >= w[i]) {
            profit += p[i];
            capacity -= w[i];
        } else {
            profit += ratio[i] * capacity;
            capacity = 0;
        }
    }

    printf("Maximum Profit = %.2f\n", profit);
}

/* ---------- 6. KRUSKAL'S ALGORITHM ---------- */
struct Edge {
    int u, v, w;
};

int findParent(int parent[], int x) {
    while (parent[x] != x) {
        parent[x] = parent[parent[x]];
        x = parent[x];
    }
    return x;
}

void kruskal(void) {
    struct Edge e[MAX], temp;
    int parent[MAXV], n, m, i, j, a, b, cost = 0, count = 0;

    printf("Enter number of vertices and edges: ");
    scanf("%d%d", &n, &m);
    if (n < 1 || n > MAXV || m < 0 || m > MAX) {
        printf("Limit: 1-%d vertices and 0-%d edges.\n", MAXV, MAX);
        return;
    }

    for (i = 0; i < n; i++) parent[i] = i;

    printf("Enter each edge as: u v weight (vertices 0 to n-1):\n");
    for (i = 0; i < m; i++) scanf("%d%d%d", &e[i].u, &e[i].v, &e[i].w);

    for (i = 0; i < m - 1; i++) {
        for (j = i + 1; j < m; j++) {
            if (e[i].w > e[j].w) {
                temp = e[i]; e[i] = e[j]; e[j] = temp;
            }
        }
    }

    for (i = 0; i < m && count < n - 1; i++) {
        a = findParent(parent, e[i].u);
        b = findParent(parent, e[i].v);

        if (a != b) {
            parent[a] = b;
            cost += e[i].w;
            count++;
            printf("%d - %d : %d\n", e[i].u, e[i].v, e[i].w);
        }
    }

    if (count == n - 1) printf("Minimum Cost = %d\n", cost);
    else printf("Spanning tree not possible\n");
}

/* ---------- 7. PRIM'S ALGORITHM ---------- */
void prim(void) {
    int n, a[MAXV][MAXV], visited[MAXV] = {0};
    int i, j, k, min, u, v, cost = 0;

    printf("Enter number of vertices: ");
    scanf("%d", &n);
    if (n < 1 || n > MAXV) {
        printf("Limit: 1-%d vertices.\n", MAXV);
        return;
    }

    printf("Enter adjacency matrix (0 means no edge):\n");
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            scanf("%d", &a[i][j]);

    visited[0] = 1;

    for (k = 0; k < n - 1; k++) {
        min = 9999; u = -1; v = -1;

        for (i = 0; i < n; i++) {
            if (visited[i]) {
                for (j = 0; j < n; j++) {
                    if (!visited[j] && a[i][j] != 0 && a[i][j] < min) {
                        min = a[i][j];
                        u = i;
                        v = j;
                    }
                }
            }
        }

        if (v == -1) {
            printf("Spanning tree not possible\n");
            return;
        }

        printf("%d - %d : %d\n", u, v, min);
        visited[v] = 1;
        cost += min;
    }

    printf("Minimum Cost = %d\n", cost);
}

/* ---------- 8. 0/1 KNAPSACK ---------- */
void zeroOneKnapsack(void) {
    int n, p[MAX + 1], w[MAX + 1], capacity;
    int dp[MAX + 1][MAXCAP + 1] = {{0}};
    int i, j, take;

    printf("Enter number of items (max %d): ", MAX);
    scanf("%d", &n);
    if (n < 0 || n > MAX) {
        printf("Invalid number of items.\n");
        return;
    }

    printf("Enter profit and weight of each item:\n");
    for (i = 1; i <= n; i++) scanf("%d%d", &p[i], &w[i]);

    printf("Enter capacity (max %d): ", MAXCAP);
    scanf("%d", &capacity);
    if (capacity < 0 || capacity > MAXCAP) {
        printf("Capacity must be between 0 and %d.\n", MAXCAP);
        return;
    }

    for (i = 1; i <= n; i++) {
        for (j = 0; j <= capacity; j++) {
            dp[i][j] = dp[i - 1][j];

            if (w[i] <= j) {
                take = p[i] + dp[i - 1][j - w[i]];
                if (take > dp[i][j]) dp[i][j] = take;
            }
        }
    }

    printf("Maximum Profit = %d\n", dp[n][capacity]);
}

/* ---------- 9. BREADTH-FIRST SEARCH (BFS) ---------- */
void bfs(void) {
    int n, a[MAXV][MAXV], visited[MAXV] = {0};
    int q[MAXV], front = 0, rear = 0;
    int i, j, start, v;

    printf("Enter number of vertices: ");
    scanf("%d", &n);
    if (n < 1 || n > MAXV) {
        printf("Limit: 1-%d vertices.\n", MAXV);
        return;
    }

    printf("Enter adjacency matrix (0 or 1):\n");
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            scanf("%d", &a[i][j]);

    printf("Enter starting vertex (0 to %d): ", n - 1);
    scanf("%d", &start);
    if (start < 0 || start >= n) {
        printf("Invalid starting vertex.\n");
        return;
    }

    visited[start] = 1;
    q[rear++] = start;

    printf("BFS Traversal: ");
    while (front < rear) {
        v = q[front++];
        printf("%d ", v);

        for (i = 0; i < n; i++) {
            if (a[v][i] == 1 && !visited[i]) {
                visited[i] = 1;
                q[rear++] = i;
            }
        }
    }
    printf("\n");
}

/* ---------- 10. DEPTH-FIRST SEARCH (DFS) ---------- */
int graph[MAXV][MAXV], dfsVisited[MAXV], vertices;

void dfsVisit(int v) {
    int i;
    printf("%d ", v);
    dfsVisited[v] = 1;

    for (i = 0; i < vertices; i++) {
        if (graph[v][i] == 1 && !dfsVisited[i])
            dfsVisit(i);
    }
}

void dfs(void) {
    int i, j, start;

    printf("Enter number of vertices: ");
    scanf("%d", &vertices);
    if (vertices < 1 || vertices > MAXV) {
        printf("Limit: 1-%d vertices.\n", MAXV);
        return;
    }

    for (i = 0; i < vertices; i++) {
        dfsVisited[i] = 0;
        for (j = 0; j < vertices; j++)
            scanf("%d", &graph[i][j]);
    }

    printf("Enter starting vertex (0 to %d): ", vertices - 1);
    scanf("%d", &start);
    if (start < 0 || start >= vertices) {
        printf("Invalid starting vertex.\n");
        return;
    }

    printf("DFS Traversal: ");
    dfsVisit(start);
    printf("\n");
}
