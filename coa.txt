CODE 2 (BOOTH MULTIPLICATION):

#include <stdio.h>

void printBinary(int n, int bits)
{
    int i;
    for (i = bits - 1; i >= 0; i--)
        printf("%d", (n >> i) & 1);
}

int main()
{
    int M, Q, A = 0, Q_1 = 0;
    int n = 5, count = 5, result;

    printf("Enter multiplicand: ");
    scanf("%d", &M);

    printf("Enter multiplier: ");
    scanf("%d", &Q);

    printf("Initial Values: A=");
    printBinary(A, n);
    printf("\tQ=");
    printBinary(Q, n);
    printf("\tQ_1=%d\n\n", Q_1);

    while (count--)
    {
        if ((Q & 1) == 1 && Q_1 == 0)
            A = A - M;
        else if ((Q & 1) == 0 && Q_1 == 1)
            A = A + M;

        printf("Step %d: A=", 5 - count);
        printBinary(A, n);
        printf("\tQ=");
        printBinary(Q, n);
        printf("\tQ_1=%d\n", Q_1);

        Q_1 = Q & 1;
        Q = (A % 2 != 0) ? (Q >> 1) | (1 << (n - 1)) : (Q >> 1);
        A = A >> 1;
    }

    result = (A << n) | Q;

    printf("Final product: ");
    printBinary(result, 2 * n);
    printf("\nAnswer: %d\n", result);

    return 0;
}

CODE 7C (REVERSE STRING):
.model small
.stack 100h
.data
str db 'REVERSE', '$'
rev db 20 dup('$')
msg1 db 'Original: $'
msg2 db 0Dh, 0Ah, 'Reversed: $'

.code
start:
    mov ax, @data
    mov ds, ax
    mov es, ax

    lea dx, msg1
    mov ah, 09h
    int 21h

    lea dx, str
    mov ah, 09h
    int 21h

    lea si, str
    mov cx, 7

push_chars:
    lodsb
    xor ah, ah
    push ax
    loop push_chars

    lea di, rev
    mov cx, 7

pop_chars:
    pop ax
    stosb
    loop pop_chars

    mov al, '$'
    stosb

    lea dx, msg2
    mov ah, 09h
    int 21h

    lea dx, rev
    mov ah, 09h
    int 21h

    mov ax, 4C00h
    int 21h
end start

CODE 8 (GREATEST NUM):
.model small
.stack 100h
.data
msg1 db 'Enter number of elements: $'
msg2 db 13,10,'Enter number: $'
msg3 db 13,10,'Greatest number is: $'
n db ?
max db ?

.code
start:
    mov ax, @data
    mov ds, ax

    lea dx, msg1
    mov ah, 09h
    int 21h
    call read_num
    mov n, al

    lea dx, msg2
    mov ah, 09h
    int 21h
    call read_num
    mov max, al

    mov cl, n
    dec cl
    jz result

next_num:
    lea dx, msg2
    mov ah, 09h
    int 21h
    call read_num

    cmp al, max
    jbe skip
    mov max, al

skip:
    loop next_num

result:
    lea dx, msg3
    mov ah, 09h
    int 21h

    mov al, max
    add al, '0'
    mov dl, al
    mov ah, 02h
    int 21h

    mov ax, 4c00h
    int 21h

read_num proc
    mov ah, 01h
    int 21h
    sub al, '0'
    ret
read_num endp

end start

CODE 9 (FILE OPERATION):
.model small
.stack 100h
.data
fname db 'TEST.TXT',0
msg db 'Hello! File created using 8086 MASM.'
buf db 100 dup('$')
m1 db 13,10,'File written successfully.$'
m2 db 13,10,'File contents: $'
m3 db 13,10,'File error!$'
handle dw ?

.code
start:
    mov ax, @data
    mov ds, ax

    ; Create file
    mov ah, 3Ch
    mov cx, 0
    lea dx, fname
    int 21h
    jc error
    mov handle, ax

    ; Write to file
    mov ah, 40h
    mov bx, handle
    mov cx, 36
    lea dx, msg
    int 21h
    jc error

    ; Close file
    mov ah, 3Eh
    mov bx, handle
    int 21h

    lea dx, m1
    mov ah, 09h
    int 21h

    ; Open file for reading
    mov ah, 3Dh
    mov al, 0
    lea dx, fname
    int 21h
    jc error
    mov handle, ax

    ; Read file
    mov ah, 3Fh
    mov bx, handle
    mov cx, 100
    lea dx, buf
    int 21h
    jc error
    mov si, ax
    mov byte ptr buf[si], '$'

    ; Display contents
    lea dx, m2
    mov ah, 09h
    int 21h
    lea dx, buf
    mov ah, 09h
    int 21h

    ; Close file
    mov ah, 3Eh
    mov bx, handle
    int 21h
    jmp exit

error:
    lea dx, m3
    mov ah, 09h
    int 21h

exit:
    mov ax, 4C00h
    int 21h
end start
